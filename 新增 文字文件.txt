    function updateProgress(percent) {
        document.getElementById('progress').style.width = percent + '%';
    }

    function nextQuestion(currentQ, answer) {
        document.getElementById(`q${currentQ}`).classList.remove('active');

        if (currentQ === 1) {
            if (answer === 'rain') {
                showResult(7);
            } else {
                document.getElementById('q2').classList.add('active');
                updateProgress(66);
            }
        } else if (currentQ === 2) {
            if (answer === 'hike') {
                document.getElementById('q3-hike').classList.add('active');
            } else {
                document.getElementById('q3-chill').classList.add('active');
            }
            updateProgress(90);
        }
    }

    function showResult(optionId) {
        const cards = document.querySelectorAll('.question-card');
        cards.forEach(card => card.classList.remove('active'));

        const res = data[optionId];
        document.getElementById('res-title').innerText = res.title;
        document.getElementById('res-tag').innerText = res.tag;
        document.getElementById('res-desc').innerText = res.desc;
        document.getElementById('res-photo').innerHTML = res.photo;

        const timelineList = document.getElementById('res-timeline');
        timelineList.innerHTML = '';
        res.timeline.forEach(item => {
            const li = document.createElement('li');
            li.innerHTML = item;
            timelineList.appendChild(li);
        });

        document.getElementById('result').style.display = 'block';
        updateProgress(100);
    }

    function resetQuiz() {
        document.getElementById('result').style.display = 'none';
        const cards = document.querySelectorAll('.question-card');
        cards.forEach(card => card.classList.remove('active'));

        document.getElementById('q1').classList.add('active');
        updateProgress(33);
    }
